        serp = fetch_serp(kw, engine='yandex', location='izhevsk')
        analyze_features(serp, kw)
        if position_changed(kw, threshold=3):
            alert_telegram(f"Position change: {kw} {old_pos}→{new_pos}")

    # Weekly competitor check
