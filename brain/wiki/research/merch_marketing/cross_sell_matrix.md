// Algorithm for dynamic recommendations
function getRecommendations(baseProduct, cartItems = []) {
  const rules = CROSS_SELL_RULES[baseProduct.sku] || CROSS_SELL_RULES[baseProduct.category];
  if (!rules) return [];
  
  // Filter out items already in cart
  const inCart = new Set(cartItems.map(item => item.sku));
  const recommendations = [];
  
  if (rules.primary && !inCart.has(rules.primary)) {
    recommendations.push({ sku: rules.primary, type: 'primary', discount: rules.discount });
  }
  
  for (const secondary of rules.secondary) {
    if (!inCart.has(secondary)) {
      recommendations.push({ sku: secondary, type: 'secondary', discount: rules.discount });
    }
  }
  
  if (rules.bundle && !cartHasBundle(cartItems, rules.bundle)) {
    recommendations.push({ bundle: rules.bundle, type: 'bundle', discount: rules.discount });
  }
  
  return recommendations.slice(0, 3); // Max 3 recommendations
}
